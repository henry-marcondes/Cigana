import Link from 'next/link';
import { apiFetch } from '../../../../../../../services/api';
import CenaConteudo from '../../../../../../../components/CenaConteudo';

const API_URL = 'http://localhost:3001';

/*
 * Converte os caminhos de mídia vindos do backend
 * para URLs utilizáveis pelo navegador.
 */
function montarUrlMedia(url) {
  if (!url) {
    return null;
  }

  if (url.startsWith('http')) {
    return url;
  }

  const caminho = url
    .replace(/^.*?media\//, 'media/')
    .replace(/^\/+/, '');

  return `${API_URL}/${caminho}`;
}

/*
 * Carrega o conteúdo do arquivo Markdown da cena.
 */
async function carregarTexto(textoUrl) {
  const url = montarUrlMedia(textoUrl);

  if (!url) {
    return '';
  }

  const resposta = await fetch(url, {
    cache: 'no-store',
  });

  if (!resposta.ok) {
    throw new Error(
      `Não foi possível carregar o texto: ${textoUrl}`
    );
  }

  return resposta.text();
}

export default async function CenaPage({ params }) {
  const { slug, capituloSlug, cenaSlug } = await params;

  /*
   * ============================================================
   * 1. LIVRO
   * ============================================================
   */

  const respostaLivro = await apiFetch(
    `/api/livros/slug/${slug}`
  );

  const livro = respostaLivro.data;

  /*
   * ============================================================
   * 2. CAPÍTULO
   * ============================================================
   */

  const respostaCapitulo = await apiFetch(
    `/api/capitulos/livro/${livro.id}/slug/${capituloSlug}`
  );

  const capitulo = respostaCapitulo.data;

  /*
   * ============================================================
   * 3. CENA
   * ============================================================
   */

  const respostaCena = await apiFetch(
    `/api/cenas/capitulo/${capitulo.id}/slug/${cenaSlug}`
  );

  const cena = respostaCena.data;

  /*
   * ============================================================
   * 4. CENA CONTEÚDO
   * ============================================================
   */

  const respostaConteudos = await apiFetch(
    `/api/cena-conteudos/cena/${cena.id}`
  );

  const cenaConteudos = respostaConteudos.data || [];

  /*
   * ============================================================
   * 5. RESOLVER OS OBJETOS DE CONTEÚDO
   * ============================================================
   */

  const conteudos = await Promise.all(
    cenaConteudos
      .filter((item) => item.ativo)
      .sort(
        (a, b) =>
          a.ordem_exibicao - b.ordem_exibicao
      )
      .map(async (item) => {
        let dados = null;

        /*
         * --------------------------------------------------------
         * TEXTO
         * --------------------------------------------------------
         */

        if (item.tipo_conteudo === 'TEXTO') {
          const resposta = await apiFetch(
            `/api/cena-textos/${item.conteudo_id}`
          );

          const texto = resposta.data;

          dados = {
            ...texto,
            texto: await carregarTexto(
              texto.texto_url
            ),
          };
        }

        /*
         * --------------------------------------------------------
         * IMAGEM
         * --------------------------------------------------------
         */

        else if (item.tipo_conteudo === 'IMAGEM') {
          const resposta = await apiFetch(
            `/api/cena-imagens/${item.conteudo_id}`
          );

          const imagem = resposta.data;

          dados = {
            ...imagem,
            imagem_url: montarUrlMedia(
              imagem.imagem_url
            ),
          };
        }

        /*
         * --------------------------------------------------------
         * ÁUDIO
         * --------------------------------------------------------
         */

        else if (item.tipo_conteudo === 'AUDIO') {
          const resposta = await apiFetch(
            `/api/cena-audios/${item.conteudo_id}`
          );

          const audio = resposta.data;

          dados = {
            ...audio,
            audio_url: montarUrlMedia(
              audio.audio_url
            ),
          };
        }

        /*
         * --------------------------------------------------------
         * VÍDEO
         * --------------------------------------------------------
         */

        else if (item.tipo_conteudo === 'VIDEO') {
          const resposta = await apiFetch(
            `/api/cena-videos/${item.conteudo_id}`
          );

          const video = resposta.data;

          dados = {
            ...video,
            video_url: montarUrlMedia(
              video.video_url
            ),
            miniatura_url: montarUrlMedia(
              video.miniatura_url
            ),
          };
        }

        return {
          ...item,
          dados,
        };
      })
  );

  /*
   * ============================================================
   * 6. ESCOLHAS
   * ============================================================
   */

  const respostaEscolhas = await apiFetch(
    `/api/escolhas/cena/${cena.id}`
  );

  const escolhas = respostaEscolhas.data || [];

  /*
   * ============================================================
   * 7. RESOLVER DESTINOS DAS ESCOLHAS
   * ============================================================
   */

  const escolhasComDestino = await Promise.all(
    escolhas.map(async (escolha) => {
      const respostaDestino = await apiFetch(
        `/api/cenas/${escolha.cena_destino_id}`
      );

      return {
        ...escolha,
        cenaDestino: respostaDestino.data,
      };
    })
  );

  /*
   * ============================================================
   * 8. RENDERIZAÇÃO
   * ============================================================
   */

  return (
    <main className="min-h-screen bg-gray-100 p-6">
      <div className="mx-auto max-w-4xl">

        <Link
          href={`/livros/${slug}/capitulos/${capituloSlug}`}
          className="mb-6 inline-block text-sm text-gray-600 hover:underline"
        >
          ← Voltar para o capítulo
        </Link>

        <article className="overflow-hidden rounded-lg bg-white shadow-md">

          {/* Cabeçalho da cena */}

          <header className="border-b border-gray-200 p-8">

            <p className="text-sm font-medium text-gray-500">
              {livro.titulo}
            </p>

            <p className="mt-1 text-sm text-gray-500">
              {capitulo.titulo}
            </p>

            <h1 className="mt-4 text-3xl font-bold text-gray-900">
              {cena.titulo}
            </h1>

          </header>

          {/* Conteúdo da cena */}

          <div className="p-8">

            {conteudos.length === 0 ? (
              <p className="text-gray-600">
                Esta cena ainda não possui conteúdo.
              </p>
            ) : (
              <div className="space-y-8">

                {conteudos.map((conteudo) => (
                  <CenaConteudo
                    key={conteudo.id}
                    conteudo={conteudo}
                  />
                ))}

              </div>
            )}

            {/* =================================================
                ESCOLHAS
                ================================================= */}

            {escolhasComDestino.length > 0 && (
              <section className="mt-10 border-t border-gray-200 pt-6">

                <h2 className="mb-4 text-xl font-semibold text-gray-900">
                  O que deseja fazer?
                </h2>

                <div className="flex flex-col gap-3">

                  {escolhasComDestino.map((escolha) => (
                    <Link
                      key={escolha.id}
                      href={`/livros/${slug}/capitulos/${capituloSlug}/cenas/${escolha.cenaDestino.slug}`}
                      className="rounded-lg bg-primary px-5 py-3 text-center font-medium text-white transition hover:opacity-90"
                    >
                      {escolha.texto}
                    </Link>
                  ))}

                </div>

              </section>
            )}

            {/* =================================================
                FINAL DA CENA
                ================================================= */}

            {escolhasComDestino.length === 0 && (
              <section className="mt-10 border-t border-gray-200 pt-6">

                <p className="text-center text-gray-600">
                  Fim desta cena.
                </p>

                <div className="mt-4 text-center">

                  <Link
                    href={`/livros/${slug}/capitulos/${capituloSlug}`}
                    className="text-sm font-medium text-primary hover:underline"
                  >
                    ← Voltar para o capítulo
                  </Link>

                </div>

              </section>
            )}

          </div>
        </article>
      </div>
    </main>
  );
}
